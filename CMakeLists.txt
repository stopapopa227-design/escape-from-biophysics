cmake_minimum_required(VERSION 3.20)
project(EscapeFromBiophysics LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This edition uses Win32/WGL. Build on Windows.")
endif()
add_executable(EscapeFromBiophysics WIN32 src/main.cpp)
target_compile_features(EscapeFromBiophysics PRIVATE cxx_std_17)
target_compile_definitions(EscapeFromBiophysics PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(EscapeFromBiophysics PRIVATE opengl32 gdi32 user32 windowscodecs ole32)
add_custom_command(TARGET EscapeFromBiophysics POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:EscapeFromBiophysics>/assets")
if(MSVC)
  target_compile_options(EscapeFromBiophysics PRIVATE /W4 /utf-8 /permissive-)
endif()
